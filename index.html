  list[i] = list[j];

Kode Program VS Code:


// Nama  : Putra Dhafir Nashrullah
// NIM   : 251410009
// Kelas : SI.2A

// =======================
// 1. STACK
// =======================
class Stack {
  final List _items = [];

  void push(T value) {
    _items.add(value);
  }

  T? pop() {
    if (_items.isEmpty) return null;
    return _items.removeLast();
  }

  T? peek() {
    if (_items.isEmpty) return null;
    return _items.last;
  }
}

// =======================
// 2. MERGE LISTS
// =======================
List mergeLists(List a, List b) {
  return [...a, ...b];
}

// =======================
// 3. RESPONSE
// =======================
class Response {
  T? data;
  String? error;

  Response({this.data, this.error});
}

// =======================
// MAIN
// =======================
void main() {
  // Stack
  var stack = Stack();
  stack.push(10);
  stack.push(20);
  print('Peek: ${stack.peek()}');
  print('Pop: ${stack.pop()}');

  // Merge Lists
  var result = mergeLists([1, 2], [3, 4]);
  print('Merge: $result');

  // Response
  var res = Response(data: "Berhasil");
  print('Response: ${res.data}');
}

Class Response merupakan generic class yang digunakan untuk menampung hasil proses, baik berupa data maupun error. Pada fungsi main(), dibuat objek Stack untuk menyimpan angka, kemudian dilakukan operasi push, pop, dan peek. Selain itu, digunakan fungsi mergeLists untuk menggabungkan dua list, serta objek Response untuk menampilkan hasil response. Dengan penggunaan generic, program menjadi lebih fleksibel, reusable, dan type-safe karena satu struktur data dan fungsi dapat digunakan untuk berbagai tipe data tanpa perubahan kode utama.

class Repository {
  final List _data = [];

  // tambah data
  void add(T item) {
    _data.add(item);
  }

  // ambil semua data
  List getAll() {
    return _data;
  }

  // hapus data
  void remove(T item) {
    _data.remove(item);
  }
}

// =======================
// CONTOH CLASS
// =======================
class Product {
  String name;
  int price;

  Product(this.name, this.price);

  @override
  String toString() {
    return 'Product(name: $name, price: $price)';
  }
}

// =======================
// MAIN
// =======================
void main() {
  // Repository untuk String
  var userRepo = Repository();
  userRepo.add("Alice");
  userRepo.add("Bob");

  print("User Repo: ${userRepo.getAll()}");

  // Repository untuk Product
  var productRepo = Repository();
  productRepo.add(Product("Laptop", 1000));
  productRepo.add(Product("Mouse", 200));

  print("Product Repo: ${productRepo.getAll()}");

  // Hapus data
  userRepo.remove("Alice");
